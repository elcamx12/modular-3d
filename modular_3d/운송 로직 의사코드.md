Enum Posture {
  LYING   = "lying"     # 눕힘
  STANDING = "standing"  # 세움 (단순 wall + A-frame 트럭 전용)
}

Enum PlacementSlot {
  FLOOR     = "floor"      # 트럭 바닥 옆자리
  STACK     = "stack"      # 위층 적층 (LYING 만)
  DEP_INNER = "dep_inner"  # 종속 패널 내부 슬롯 (LYING 만)
}

@dataclass(frozen=True)
Placement {
  item:        Module | Panel
  slot:        PlacementSlot
  posture:     Posture
  truck_xyz:   tuple[float, float, float]  # 트럭 적재함 중심 원점 (mm)
  parent_idx:  int | None  # STACK / DEP_INNER 일 때 모체 Placement 의 trip.placements 내 인덱스
}

@dataclass
TruckState {
  truck:                 Truck
  effective_cargo_limit: float  # = _effective_cargo_limit(truck, site)
  used_floor_length:     float  # 누적 점유 길이 (mm)
  used_cargo_weight:     float  # 누적 화물 무게 (kg)
  layer_max_thickness:   list[float]  # 단별 max 두께 (mm) — len = 단 수
  placements:            list[Placement]  # 이 트럭에 배치된 화물 목록

  # 파생 속성
  property remaining_cargo_limit -> float:
      return self.effective_cargo_limit − self.used_cargo_weight
  property remaining_floor_length(usable_length) -> float:
      return usable_length − self.used_floor_length
  property used_inner_height(panel_gap_mm) -> float:
      n = len(self.layer_max_thickness)
      if n == 0: return 0.0
      return sum(self.layer_max_thickness) + max(n − 1, 0) * panel_gap_mm
}

@dataclass(frozen=True)
CheckResult {
  ok:      bool
  reasons: tuple[str, ...]
}

function _item_dims_for_posture(item, posture) -> (length, width, height_occupied, weight):
    # item 의 점유 차원을 자세에 따라 변환
    # - Module: 항상 (length, width, height, weight)
    # - Panel LYING: (length, width, _dependent_max_thickness(item) or thickness, weight)
    # - Panel STANDING: (length, thickness, width, weight)

function can_place(item, posture, slot, truck_state, site, sp) -> CheckResult:
    length, width, height_occupied, weight = _item_dims_for_posture(item, posture)
    tr = truck_state.truck

    reasons = []
    # ① 길이
    usable = tr.max_length − 2 × sp.truck_edge_clearance_mm
    gap = sp.panel_gap_mm if truck_state.placements else 0
    if truck_state.used_floor_length + gap + length > usable:
        reasons.append(f"길이 초과 …")

    # ② 폭
    eff_w = tr.max_width + 2 × sp.side_overhang_mm
    if width > eff_w:
        reasons.append(f"폭 초과 (트럭) …")
    if site.max_width_mm is not None and width > site.max_width_mm:
        reasons.append(f"폭 초과 (현장) …")

    # ③ 높이 — slot 에 따라
    inner_h = tr.max_height − tr.vehicle_height_offset
    new_layer_top = (truck_state.used_inner_height(sp.panel_gap_mm) if slot == STACK else 0)
    if new_layer_top + height_occupied > inner_h:
        reasons.append(f"내공높이 초과 …")
    if site.max_height_mm is not None:
        outer = (new_layer_top + height_occupied) + tr.vehicle_height_offset
        if outer > site.max_height_mm:
            reasons.append(f"높이 초과 (현장) …")

    # ④ 무게 — 실효 화물 한도 (현장 GVW 누적 반영)
    if truck_state.used_cargo_weight + weight > truck_state.effective_cargo_limit:
        reasons.append(f"중량 초과 …")

    return CheckResult(ok=not reasons, reasons=tuple(reasons))
```

### 2.4 자리 평가 (`packer_core.py`)

```
function evaluate_slot(item, posture, slot, truck_state, weights, site, sp) -> tuple[float, dict]:
    """자리 평가 점수 반환. 미통과면 (-∞, None) 반환.

    [사용자 결정 — 정확 모드]
    - 길이 분모: 잔여 적재 공간 (유효 길이 − 깔린 패널 길이 합 − gap 합 − 새 gap)
    - 높이 점수: 모든 자리에서 계산 (바닥에 두꺼운 거 놓는 손해도 반영)
    """
    check = can_place(item, posture, slot, truck_state, site, sp)
    if not check.ok:
        return float("-inf"), None
    α, β, γ = weights
    length, width, h_occ, w_kg = _item_dims_for_posture(item, posture)
    tr = truck_state.truck

    # ① 길이 활용도 — 잔여 적재 공간 기준 (사용자 결정)
    usable = tr.max_length − 2 * sp.truck_edge_clearance_mm
    gap_count = max(0, len(truck_state.placements))  # 기존 패널 수 만큼 gap
    used_with_gap = truck_state.used_floor_length + gap_count * sp.panel_gap_mm
    new_gap = sp.panel_gap_mm if truck_state.placements else 0
    remaining_len = max(1.0, usable − used_with_gap − new_gap)  # 0 나누기 방지
    Δlen = length / remaining_len if slot == FLOOR else length / usable
        # STACK/DEP_INNER 는 바닥 길이 안 쓰니 유효 길이 분모

    # ② 무게 활용도 — 잔여 화물 한도 기준 (이미 누적 차감)
    remaining_wt = max(1.0, truck_state.remaining_cargo_limit)
    Δwt = w_kg / remaining_wt

    # ③ 높이 활용도 — 모든 자리에서 계산 (사용자 결정 — 라)
    # 바닥/슬롯에 두꺼운 패널 놓으면 위에 쌓을 공간 줄어드는 손해 반영
    inner_h = tr.max_height − tr.vehicle_height_offset
    if slot == STACK:
        # 적층 자리는 새 단 두께가 그대로 높이 점수
        Δht = h_occ / inner_h
    elif slot == FLOOR:
        # 바닥 자리도 *그 패널의 두께만큼 위 공간 차감* 영향 반영
        Δht = h_occ / inner_h
    else:  # DEP_INNER — 종속 슬롯
        Δht = h_occ / inner_h

    score = α * Δlen + β * Δwt + γ * Δht

    # STACK 기하 제약 검사 (위 단 ≤ 아래 단)
    if slot == STACK:
        bottom_idx = _find_stack_bottom(truck_state)
        if bottom_idx is None: return float("-inf"), None
        bottom = truck_state.placements[bottom_idx]
        if length > bottom.item_length or width > bottom.item_width:
            return float("-inf"), None

    # DEP_INNER 슬롯 검사 (벽 사이 빈 공간)
    if slot == DEP_INNER:
        free_w, free_l = _dependent_free_inner_dims(parent_panel, sp)
        if width > free_w or length > free_l: return float("-inf"), None

    return score, {"slot": slot, "posture": posture,
                   "xyz": _compute_xyz(truck_state, slot, posture, item, sp)}


function _compute_xyz(truck_state, slot, posture, item, sp) -> tuple[float,float,float]:
    """트럭 적재함 중심 원점 좌표. 적재 끝 위치를 자리·자세별로 계산."""
    tr = truck_state.truck
    usable = tr.max_length − 2 * sp.truck_edge_clearance_mm
    L_eff, W_eff, _, _ = _item_dims_for_posture(item, posture)

    if slot == FLOOR:
        # 트럭 좌측 끝 + 누적 길이 + gap + 패널 절반 (중앙 정렬)
        x_left_edge = −usable / 2  # 적재함 중심 원점에서 좌측 끝
        gap = sp.panel_gap_mm if truck_state.placements else 0
        x = x_left_edge + truck_state.used_floor_length + gap + L_eff / 2
        y = 0  # 트럭 폭 중앙 (보정 단계에서 미세 슬라이드)
        z = 0  # 적재함 바닥 (위층 적층이 아니므로)
    elif slot == STACK:
        bottom = truck_state.placements[_find_stack_bottom(truck_state)]
        x = bottom.truck_xyz[0]
        y = bottom.truck_xyz[1]
        z = truck_state.used_inner_height(sp.panel_gap_mm)
    elif slot == DEP_INNER:
        # 종속 패널의 빈 슬롯 중심에 배치
        parent = truck_state.placements[_find_dep_inner_parent(truck_state, item)]
        x = parent.truck_xyz[0]; y = parent.truck_xyz[1]
        z = parent.item_thickness + sp.lshape_stack_gap_mm
    return (x, y, z)
```

### 2.5 단일 패킹 — `pack_one_seed` (`packer_core.py`)

```
function pack_one_seed(items, trucks, site, sp, strategy: GreedyStrategy,
                       weights: tuple[float,float,float],
                       cost_mode: str, eco_options) -> PackResult:
    """그리디 전략 + 가중치 1 조합으로 패킹 → 1 후보 해 반환."""

    sorted_items = strategy.sort_items_fn(items)
    truck_states: list[TruckState] = []  # 활성 회차들
    blocked: list = []

    for item in sorted_items:
        # 자세 후보 결정 — isinstance + 단순 wall 만 STANDING 가능
        postures = [Posture.LYING]
        if isinstance(item, Panel) and item.kind == "wall" and not item.wall_segments:
            postures.append(Posture.STANDING)
        # 모듈·종속 floor·단순 floor 는 항상 LYING

        best_score = float("-inf")
        best_choice = None  # (truck_state, slot, posture, xyz)

        # 모든 트럭 × 자세 × 자리 후보 탐색 (Best Fit)
        for ts in truck_states:
            for posture in postures:
                # 트럭-자세 호환성
                if posture == STANDING and ts.truck.truck_type != "aframe": continue
                if posture == LYING and ts.truck.truck_type == "aframe": continue
                # A-frame 트럭은 *단순 wall STANDING* 만 받음 — 모듈·floor·종속 거부
                if ts.truck.truck_type == "aframe":
                    if not (isinstance(item, Panel) and item.kind == "wall" and not item.wall_segments):
                        continue

                allowed_slots = _allowed_slots_for(item, posture)
                # LYING: [FLOOR, STACK, DEP_INNER]
                # STANDING: [FLOOR] only
                for slot in allowed_slots:
                    score, meta = evaluate_slot(item, posture, slot, ts, weights, site, sp)
                    if score > best_score:
                        best_score = score
                        best_choice = (ts, slot, posture, meta)

        if best_choice is not None:
            ts, slot, posture, meta = best_choice
            _commit_placement(ts, item, slot, posture, meta, sp)
            continue

        # 어디에도 못 들어감 → 새 트럭 추가
        remaining = _remaining_items(sorted_items, current_index)
        new_truck = _select_best_new_truck(item, trucks, remaining, strategy, site, sp,
                                          cost_mode, eco_options)
        if new_truck is None:
            blocked.append((item, _diagnose_blocked(item, trucks, site)))
            continue
        ts_new = TruckState(truck=new_truck, ...)
        truck_states.append(ts_new)
        # 새 트럭에 모든 자세 × FLOOR 자리 다시 평가 — 자세 first only 버그 방지
        best_new = None; best_new_score = float("-inf")
        for posture in postures:
            if posture == STANDING and new_truck.truck_type != "aframe": continue
            if posture == LYING and new_truck.truck_type == "aframe": continue
            score, meta = evaluate_slot(item, posture, FLOOR, ts_new, weights, site, sp)
            if score > best_new_score:
                best_new_score = score; best_new = (posture, meta)
        if best_new is None:
            blocked.append((item, "새 트럭 추가했으나 자세 매칭 실패"))
            continue
        _commit_placement(ts_new, item, FLOOR, best_new[0], best_new[1], sp)

    # 모듈 합산 4.5 m 후처리 — 두 모듈 회차 합치기
    truck_states = _merge_small_modules(truck_states, site, sp)

    # PackResult 조립
    trips = [_truck_state_to_trip(ts, trip_no=i+1) for i, ts in enumerate(truck_states)]
    return PackResult(trips=trips, blocked=blocked)
```

### 2.6 다중 시드 + Lower Bound — `packer_meta.py`

```
DEFAULT_RANDOM_SEED_BASE = 42
N_RANDOM_SEEDS = 30

# 무작위 시드 전용 — 입력 순서 유지 (셔플 무효화 방지)
SHUFFLED_STRATEGY = GreedyStrategy(
    name="shuffled",
    sort_items_fn=lambda items: items,  # identity — 셔플 순서 그대로
    truck_score_fn=default_truck_score,
)

function pack_all_seeds(items, trucks, site, sp,
                        cost_mode: str, eco_options,  # 함수 시그니처 — cost_mode 는 문자열, 값은 eco_options
                        seed_base=DEFAULT_RANDOM_SEED_BASE,
                        random_count=N_RANDOM_SEEDS) -> tuple[PackResult, dict]:
    """54 시드 돌리고 비용 최저 채택. 디버그 메타 함께 반환."""

    lb = lower_bound_cost(items, trucks, site, sp, cost_mode, eco_options)
    candidates: list[tuple[PackResult, str]] = []
    early_exit = False

    # 24 결정론 시드 (전략 4 × 가중치 6)
    for strategy in GREEDY_STRATEGIES:
        if early_exit: break
        for weights in WEIGHT_SETS:
            res = pack_one_seed(items, trucks, site, sp, strategy, weights, cost_mode, eco_options)
            res = vnd_improve(res, cost_mode, eco_options, max_iter=None)  # 무한 (기본)
            candidates.append((res, f"{strategy.name}+{weights}"))
            # 하한 도달 → 모든 외부 루프 즉시 종료
            if compute_cost(res, cost_mode, eco_options) <= lb:
                early_exit = True
                break

    # 30 무작위 시드 — 입력 순서 유지 전용 전략 사용
    if not early_exit:
        for i in range(random_count):
            seed_i = seed_base + i
            rng = random.Random(seed_i)
            shuffled = items.copy(); rng.shuffle(shuffled)
            res = pack_one_seed(shuffled, trucks, site, sp,
                                SHUFFLED_STRATEGY,  # identity 정렬 — 셔플 순서 그대로
                                WEIGHT_SETS[0], cost_mode, eco_options)
            res = vnd_improve(res, cost_mode, eco_options, max_iter=None)
            candidates.append((res, f"random_{seed_i}"))
            if compute_cost(res, cost_mode, eco_options) <= lb:
                early_exit = True; break

    # 최저 비용 채택
    best, best_label = min(candidates, key=lambda x: compute_cost(x[0], cost_mode, eco_options))
    debug_meta = {"all_costs": [...], "best_label": best_label, "lb": lb, "early_exit": early_exit}
    return best, debug_meta


function lower_bound_cost(items, trucks, site, sp,
                         cost_mode: str, eco_options) -> float:
    """모드별 비용 하한. cost_mode 는 문자열 식별자, 실제 값은 eco_options 에서 추출."""
    panels = [it for it in items if isinstance(it, Panel)]
    modules = [it for it in items if isinstance(it, Module)]

    max_eff_cap = max(_effective_cargo_limit(t, site) for t in trucks)
    max_usable_len = max(t.max_length − 2*sp.truck_edge_clearance_mm for t in trucks)

    panel_weight_lb = ceil(sum(p.weight for p in panels) / max_eff_cap) if panels else 0
    panel_len_lb    = ceil(sum(p.length for p in panels) / max_usable_len) if panels else 0
    panel_trip_lb   = max(panel_weight_lb, panel_len_lb)

    N_module = len(modules)
    N_small = sum(1 for m in modules if m.length <= 4500)
    module_trip_lb = (N_module − N_small) + ceil(N_small / 2)

    total_trip_lb = panel_trip_lb + module_trip_lb

    if cost_mode == "fixed_per_trip":
        return total_trip_lb * eco_options.fixed_per_trip_rate
    elif cost_mode == "freight_table":
        return total_trip_lb * min(eco_options.lowbed_fixed_krw,
                                    eco_options.extendable_fixed_krw,
                                    eco_options.aframe_fixed_krw)
    elif cost_mode == "per_km":
        # 트럭별 km단가가 다르면 min 사용; 동일 단가면 자동으로 동일값
        rates = [eco_options.km_rate_for_truck(t) for t in trucks]
        return total_trip_lb * eco_options.round_trip_km * min(rates)
    else:
        raise ValueError(f"unknown cost_mode: {cost_mode}")
```

### 2.7 VND 국소 탐색 — `packer_local_search.py`

```
function vnd_improve(pack: PackResult, cost_mode, eco_options, max_iter=None) -> PackResult:
    """VND — 작은 수술 → 큰 수술 순서로 적용, 큰 수술 개선 시 작은 수술로 복귀."""

    operations = [
        op_transfer,       # 1. 얕음
        op_swap,           # 2. 중간
        op_posture_toggle, # 3. 수직 변화
        op_merge,          # 4. 깊음
        op_truck_downgrade,# 5. 수직
    ]
    iter_count = 0
    op_idx = 0
    while op_idx < len(operations):
        if max_iter is not None and iter_count >= max_iter:

        elif not improved:
            op_idx += 1
        # improved && op_idx == 0 → 같은 수술 더 시도

    return pack

function _panel_local_cog(panel: Panel) -> tuple[float, float]:
    """계획서 § 8.4 정확 무게중심 — 동음이의 변수 분리."""
    if not panel.wall_segments:
        return (panel.width / 2, panel.length / 2)

    # 바닥 무게
    w_floor = _floor_frame_weight(panel) + panel.extra_weight_kg
    cog_floor = (panel.width / 2, panel.length / 2)
    weighted_sum = (w_floor * cog_floor[0], w_floor * cog_floor[1])
    total_w = w_floor

    for seg in panel.wall_segments:
        w_s = seg_beam_weight(seg) + 2 * seg_column_weight(seg)
        if seg.side == 0:
            xs = seg.start_offset_mm + seg.length_mm / 2
            ys = seg.thickness_mm / 2
        elif seg.side == 1:
            xs = panel.width − seg.thickness_mm / 2
            ys = seg.start_offset_mm + seg.length_mm / 2
        elif seg.side == 2:
            xs = seg.start_offset_mm + seg.length_mm / 2
            ys = panel.length − seg.thickness_mm / 2
        elif seg.side == 3:
            xs = seg.thickness_mm / 2
            ys = seg.start_offset_mm + seg.length_mm / 2
        weighted_sum = (weighted_sum[0] + w_s * xs, weighted_sum[1] + w_s * ys)
        total_w += w_s

    return (weighted_sum[0] / total_w, weighted_sum[1] / total_w)

### 2.11 트럭 선정 — 정확 시뮬레이션 (사용자 결정)

```
function default_truck_score(truck, remaining_items, site, sp, cost_mode, eco_options) -> float:
    """비용 ÷ 예상 실릴 무게. 예상 무게 = 진짜 mini-패킹 결과.

    [사용자 결정 — 정확 모드 (A)]
    후보 트럭 한 대에 남은 화물을 *진짜로* 그리디 패킹해서 실리는 무게 합산.
    시간 부담 약 2~5배. 결과 신뢰도 ↑.
    """
    # 1회 고정비 (모드별)
    if cost_mode == "fixed_per_trip":
        fixed = eco_options.fixed_per_trip_rate
    elif cost_mode == "freight_table":
        fixed = eco_options.fixed_rate_for_truck_type(truck.truck_type)
    elif cost_mode == "per_km":
        fixed = eco_options.km_rate_for_truck(truck) * eco_options.round_trip_km
    else:
        fixed = 1.0  # 알 수 없는 모드 — 모든 트럭 동일 취급

    # 예상 실릴 무게 — mini-패킹
    expected_weight = _simulate_packing_weight(truck, remaining_items, site, sp)

    if expected_weight <= 0:
        return float("inf")  # 이 트럭에 아무것도 못 실음 → 후보에서 제외
    return fixed / expected_weight


function _simulate_packing_weight(truck, items, site, sp) -> float:
    """후보 트럭 1 대에 items 를 무게 내림차순으로 simple-FFD 적재.
    실리는 화물 무게 합 반환. 트럭 한도 차면 종료. *비파괴* — items 변경 X.
    """
    # 무게 내림차순
    sorted_items = sorted(items, key=lambda x: -x.weight)
    ts = TruckState(truck=truck, effective_cargo_limit=_effective_cargo_limit(truck, site))
    for item in sorted_items:
        # FLOOR 자리에 자세 LYING 으로 시도 (간이 — 적층 무시)
        postures = [Posture.LYING]
        if isinstance(item, Panel) and item.kind == "wall" and not item.wall_segments:
            if truck.truck_type == "aframe":
                postures = [Posture.STANDING]
        elif truck.truck_type == "aframe":
            continue  # A-frame 은 단순 wall STANDING 만
        for posture in postures:
            if can_place(item, posture, PlacementSlot.FLOOR, ts, site, sp).ok:
                # 임시 commit — TruckState 만 업데이트, Placement 는 메모리만
                length, width, h_occ, w_kg = _item_dims_for_posture(item, posture)
                ts.used_floor_length += length + (sp.panel_gap_mm if ts.placements else 0)
                ts.used_cargo_weight += w_kg
                ts.placements.append("dummy")  # gap 카운트용
                break
    return ts.used_cargo_weight
```
